# Environment-agnostic Terraform payload synthesized upon coordinate lock.
terraform {
  required_providers {
    azurerm = { source = "hashicorp/azurerm", version = "~> 3.0" }
    azuread = { source = "hashicorp/azuread", version = "~> 3.9.0" }
    github  = { source = "integrations/github", version = "~> 6.0" }
    random  = { source = "hashicorp/random", version = "~> 3.5.0" }
  }
}

provider "azurerm" { 
  features {} 
}

provider "azuread" {}
provider "github" {} 
provider "random" {}

data "azuread_client_config" "current" {}

data "github_user" "target_owner" { username = "Compcode1" }
data "github_repository" "target_repo" { full_name = "Compcode1/iteration-3" }

resource "azuread_application" "acphf_agent" {
  display_name     = "acphf-oidc-agent"
  sign_in_audience = "AzureADMyOrg"
}

resource "azuread_service_principal" "acphf_agent_sp" {
  client_id    = azuread_application.acphf_agent.client_id
  use_existing = false
}

output "ial_tenant_id" { value = data.azuread_client_config.current.tenant_id }
output "ial_application_client_id" { value = azuread_application.acphf_agent.client_id }

resource "azuread_application_federated_identity_credential" "oidc" {
  application_id = azuread_application.acphf_agent.id
  display_name   = "github-actions-federation"
  audiences      = ["api://AzureADTokenExchange"]
  issuer         = "https://token.actions.githubusercontent.com" 
  subject        = "repo:${data.github_user.target_owner.username}@${data.github_user.target_owner.id}/${data.github_repository.target_repo.name}@${data.github_repository.target_repo.repo_id}:ref:refs/heads/main"
}

data "azurerm_resource_group" "target_rg" { name = "rg-efm-test-lab-04" }

data "azurerm_key_vault" "target_vault" {
  name                = "kv-efm-test-lab-04"
  resource_group_name = data.azurerm_resource_group.target_rg.name
}

resource "azurerm_role_assignment" "kv_secrets_user" {
  scope                            = data.azurerm_key_vault.target_vault.id
  role_definition_name             = "Key Vault Secrets User"
  principal_id                     = azuread_service_principal.acphf_agent_sp.object_id
  skip_service_principal_aad_check = true
}

resource "azurerm_log_analytics_workspace" "efm_audit" {
  name                = "law-efm-audit"
  location            = data.azurerm_resource_group.target_rg.location
  resource_group_name = data.azurerm_resource_group.target_rg.name
  sku                 = "PerGB2018"
  retention_in_days   = 30
}

resource "random_string" "diag_suffix" {
  length  = 4
  special = false
  upper   = false
}

resource "azurerm_monitor_diagnostic_setting" "audit_trap" {
  name                       = "diag-kv-audit-${random_string.diag_suffix.result}"
  target_resource_id         = data.azurerm_key_vault.target_vault.id
  log_analytics_workspace_id = azurerm_log_analytics_workspace.efm_audit.id
  
  enabled_log { 
    category = "AuditEvent" 
  }
  
  metric { 
    category = "AllMetrics"
    enabled  = true 
  }
